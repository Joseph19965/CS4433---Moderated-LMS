cmake_minimum_required(VERSION 3.16)
project(CS4433_LMS VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LMS_COVERAGE "Enable GCC coverage" OFF)
option(LMS_CATCH2 "Build Catch2 tests when Catch2 is available" ON)
option(LMS_FETCH_CATCH2 "Fetch Catch2 v3 if not installed (requires internet)" OFF)
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(lms_core src/lms.cpp)
target_include_directories(lms_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(lms_core PUBLIC SQLite::SQLite3 OpenSSL::Crypto)
target_compile_options(lms_core PRIVATE -Wall -Wextra -Wpedantic)
if(LMS_COVERAGE AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_compile_options(lms_core PUBLIC --coverage -O0 -g)
  target_link_options(lms_core PUBLIC --coverage)
endif()
add_executable(lms src/main.cpp)
target_link_libraries(lms PRIVATE lms_core)
enable_testing()
add_executable(lms_tests tests/test_core.cpp)
target_link_libraries(lms_tests PRIVATE lms_core)
add_test(NAME lms_core_tests COMMAND lms_tests)
if(LMS_CATCH2)
  find_package(Catch2 3 QUIET)
  if(NOT Catch2_FOUND AND LMS_FETCH_CATCH2)
    include(FetchContent)
    FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git GIT_TAG v3.8.1 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(Catch2)
  endif()
  if(TARGET Catch2::Catch2WithMain)
    add_executable(lms_catch_tests tests/test_catch2.cpp)
    target_link_libraries(lms_catch_tests PRIVATE lms_core Catch2::Catch2WithMain)
    include(CTest)
    if(Catch2_FOUND)
      include(Catch)
    else()
      list(APPEND CMAKE_MODULE_PATH ${catch2_SOURCE_DIR}/extras)
      include(Catch)
    endif()
    catch_discover_tests(lms_catch_tests)
  else()
    message(STATUS "Catch2 v3 not installed: built-in CTest tests are available; install Catch2 v3 for additional framework tests")
  endif()
endif()
